# ROSA Terraform Starter Template
# Generated from IDPP analysis of ROSA Complete Guide
# This template provides a foundation for ROSA cluster deployment

terraform {
  required_version = ">= 1.0"
  required_providers {
    aws = {
      source  = "hashicorp/aws"
      version = "~> 5.0"
    }
    rhcs = {
      source  = "terraform-redhat/rhcs"
      version = ">= 1.6.2"
    }
  }
}

# Variables based on ROSA infrastructure analysis
variable "cluster_name" {
  description = "Name of the ROSA cluster"
  type        = string
  default     = "rosa-cluster"
}

variable "aws_region" {
  description = "AWS region for ROSA deployment"
  type        = string
  default     = "us-east-1"
}

variable "availability_zones" {
  description = "Availability zones for multi-AZ deployment"
  type        = list(string)
  default     = ["us-east-1a", "us-east-1b", "us-east-1c"]
}

variable "vpc_cidr" {
  description = "CIDR block for VPC"
  type        = string
  default     = "10.0.0.0/16"
}

variable "private_subnet_cidrs" {
  description = "CIDR blocks for private subnets"
  type        = list(string)
  default     = ["10.0.1.0/24", "10.0.2.0/24", "10.0.3.0/24"]
}

variable "public_subnet_cidrs" {
  description = "CIDR blocks for public subnets"
  type        = list(string)
  default     = ["10.0.101.0/24", "10.0.102.0/24", "10.0.103.0/24"]
}

variable "environment" {
  description = "Environment name (dev, staging, prod)"
  type        = string
  default     = "dev"
}

variable "openshift_version" {
  description = "OpenShift version for ROSA cluster"
  type        = string
  default     = "4.14"
}

variable "multi_az" {
  description = "Deploy cluster across multiple availability zones"
  type        = bool
  default     = true
}

variable "worker_node_count" {
  description = "Number of worker nodes"
  type        = number
  default     = 3
}

variable "control_plane_instance_type" {
  description = "Instance type for control plane nodes"
  type        = string
  default     = "m5.2xlarge"
}

variable "infrastructure_instance_type" {
  description = "Instance type for infrastructure nodes"
  type        = string
  default     = "r5.xlarge"
}

variable "worker_instance_type" {
  description = "Instance type for worker nodes"
  type        = string
  default     = "m5.xlarge"
}

variable "enable_fips" {
  description = "Enable FIPS mode for cluster"
  type        = bool
  default     = false
}

variable "enable_etcd_encryption" {
  description = "Enable additional etcd encryption"
  type        = bool
  default     = false
}

variable "tags" {
  description = "Tags to apply to all resources"
  type        = map(string)
  default = {
    Project     = "ROSA"
    Environment = "dev"
    ManagedBy   = "Terraform"
  }
}

# Local values for resource naming and configuration
locals {
  account_role_prefix    = "${var.cluster_name}-account"
  operator_role_prefix   = "${var.cluster_name}-operator"
  cluster_name_with_env  = "${var.cluster_name}-${var.environment}"
  
  common_tags = merge(var.tags, {
    ClusterName = var.cluster_name
    Region      = var.aws_region
  })
}

# Data sources for AWS information
data "aws_availability_zones" "available" {
  state = "available"
}

data "aws_caller_identity" "current" {}

# VPC Configuration
resource "aws_vpc" "rosa_vpc" {
  cidr_block           = var.vpc_cidr
  enable_dns_hostnames = true
  enable_dns_support   = true

  tags = merge(local.common_tags, {
    Name = "${var.cluster_name}-vpc"
    Type = "ROSA VPC"
  })
}

# Internet Gateway
resource "aws_internet_gateway" "rosa_igw" {
  vpc_id = aws_vpc.rosa_vpc.id

  tags = merge(local.common_tags, {
    Name = "${var.cluster_name}-igw"
  })
}

# Public Subnets
resource "aws_subnet" "public_subnets" {
  count = length(var.public_subnet_cidrs)

  vpc_id                  = aws_vpc.rosa_vpc.id
  cidr_block              = var.public_subnet_cidrs[count.index]
  availability_zone       = var.availability_zones[count.index]
  map_public_ip_on_launch = true

  tags = merge(local.common_tags, {
    Name = "${var.cluster_name}-public-subnet-${count.index + 1}"
    Type = "Public"
  })
}

# Private Subnets
resource "aws_subnet" "private_subnets" {
  count = length(var.private_subnet_cidrs)

  vpc_id            = aws_vpc.rosa_vpc.id
  cidr_block        = var.private_subnet_cidrs[count.index]
  availability_zone = var.availability_zones[count.index]

  tags = merge(local.common_tags, {
    Name = "${var.cluster_name}-private-subnet-${count.index + 1}"
    Type = "Private"
  })
}

# Elastic IPs for NAT Gateways
resource "aws_eip" "nat_eips" {
  count = length(aws_subnet.public_subnets)

  domain = "vpc"
  depends_on = [aws_internet_gateway.rosa_igw]

  tags = merge(local.common_tags, {
    Name = "${var.cluster_name}-nat-eip-${count.index + 1}"
  })
}

# NAT Gateways
resource "aws_nat_gateway" "nat_gateways" {
  count = length(aws_subnet.public_subnets)

  allocation_id = aws_eip.nat_eips[count.index].id
  subnet_id     = aws_subnet.public_subnets[count.index].id

  tags = merge(local.common_tags, {
    Name = "${var.cluster_name}-nat-gw-${count.index + 1}"
  })

  depends_on = [aws_internet_gateway.rosa_igw]
}

# Route Table for Public Subnets
resource "aws_route_table" "public_rt" {
  vpc_id = aws_vpc.rosa_vpc.id

  route {
    cidr_block = "0.0.0.0/0"
    gateway_id = aws_internet_gateway.rosa_igw.id
  }

  tags = merge(local.common_tags, {
    Name = "${var.cluster_name}-public-rt"
  })
}

# Route Tables for Private Subnets
resource "aws_route_table" "private_rt" {
  count = length(aws_subnet.private_subnets)

  vpc_id = aws_vpc.rosa_vpc.id

  route {
    cidr_block     = "0.0.0.0/0"
    nat_gateway_id = aws_nat_gateway.nat_gateways[count.index].id
  }

  tags = merge(local.common_tags, {
    Name = "${var.cluster_name}-private-rt-${count.index + 1}"
  })
}

# Route Table Associations
resource "aws_route_table_association" "public_rta" {
  count = length(aws_subnet.public_subnets)

  subnet_id      = aws_subnet.public_subnets[count.index].id
  route_table_id = aws_route_table.public_rt.id
}

resource "aws_route_table_association" "private_rta" {
  count = length(aws_subnet.private_subnets)

  subnet_id      = aws_subnet.private_subnets[count.index].id
  route_table_id = aws_route_table.private_rt[count.index].id
}

# S3 VPC Endpoint for container registry
resource "aws_vpc_endpoint" "s3_endpoint" {
  vpc_id       = aws_vpc.rosa_vpc.id
  service_name = "com.amazonaws.${var.aws_region}.s3"
  
  route_table_ids = concat(
    [aws_route_table.public_rt.id],
    aws_route_table.private_rt[*].id
  )

  tags = merge(local.common_tags, {
    Name = "${var.cluster_name}-s3-endpoint"
  })
}

# Security Groups for ROSA cluster components
resource "aws_security_group" "rosa_control_plane_sg" {
  name_prefix = "${var.cluster_name}-control-plane-"
  vpc_id      = aws_vpc.rosa_vpc.id
  description = "Security group for ROSA control plane nodes"

  # API server access
  ingress {
    from_port   = 6443
    to_port     = 6443
    protocol    = "tcp"
    cidr_blocks = [var.vpc_cidr]
    description = "Kubernetes API server"
  }

  # Machine config server
  ingress {
    from_port   = 22623
    to_port     = 22623
    protocol    = "tcp"
    cidr_blocks = [var.vpc_cidr]
    description = "Machine config server"
  }

  # SSH access
  ingress {
    from_port   = 22
    to_port     = 22
    protocol    = "tcp"
    cidr_blocks = [var.vpc_cidr]
    description = "SSH access"
  }

  # All outbound traffic
  egress {
    from_port   = 0
    to_port     = 0
    protocol    = "-1"
    cidr_blocks = ["0.0.0.0/0"]
    description = "All outbound traffic"
  }

  tags = merge(local.common_tags, {
    Name = "${var.cluster_name}-control-plane-sg"
    Type = "ControlPlane"
  })
}

resource "aws_security_group" "rosa_worker_sg" {
  name_prefix = "${var.cluster_name}-worker-"
  vpc_id      = aws_vpc.rosa_vpc.id
  description = "Security group for ROSA worker nodes"

  # SSH access
  ingress {
    from_port   = 22
    to_port     = 22
    protocol    = "tcp"
    cidr_blocks = [var.vpc_cidr]
    description = "SSH access"
  }

  # NodePort services
  ingress {
    from_port   = 30000
    to_port     = 32767
    protocol    = "tcp"
    cidr_blocks = [var.vpc_cidr]
    description = "NodePort services"
  }

  # All outbound traffic
  egress {
    from_port   = 0
    to_port     = 0
    protocol    = "-1"
    cidr_blocks = ["0.0.0.0/0"]
    description = "All outbound traffic"
  }

  tags = merge(local.common_tags, {
    Name = "${var.cluster_name}-worker-sg"
    Type = "Worker"
  })
}

# S3 bucket for ROSA image registry
resource "aws_s3_bucket" "rosa_registry" {
  bucket = "${var.cluster_name}-${var.environment}-registry-${random_id.bucket_suffix.hex}"

  tags = merge(local.common_tags, {
    Name = "${var.cluster_name}-registry"
    Type = "ImageRegistry"
  })
}

resource "random_id" "bucket_suffix" {
  byte_length = 4
}

resource "aws_s3_bucket_versioning" "rosa_registry_versioning" {
  bucket = aws_s3_bucket.rosa_registry.id
  versioning_configuration {
    status = "Enabled"
  }
}

resource "aws_s3_bucket_encryption" "rosa_registry_encryption" {
  bucket = aws_s3_bucket.rosa_registry.id

  server_side_encryption_configuration {
    rule {
      apply_server_side_encryption_by_default {
        sse_algorithm = "AES256"
      }
    }
  }
}

resource "aws_s3_bucket_public_access_block" "rosa_registry_pab" {
  bucket = aws_s3_bucket.rosa_registry.id

  block_public_acls       = true
  block_public_policy     = true
  ignore_public_acls      = true
  restrict_public_buckets = true
}
